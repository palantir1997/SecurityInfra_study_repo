#!/bin/bash

# ====================================================================
# 프로그램명: Ubuntu 24.04 시스템 취약점 점검 스크립트
# 작성 목적: 서버의 보안 취약점을 자동으로 진단하고 취약점을 리포팅함
# ====================================================================

# [색상 변수 정의] 결과 출력을 가독성 있게 만들기 위해 ANSI 이스케이프 코드 사용
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color (색상 초기화)

echo "====================================================="
echo "        Ubuntu 24.04 보안 취약점 점검을 시작합니다.   "
echo "====================================================="
echo "점검 일시: $(date)"
echo "----------------------------------------------------"

# --------------------------------------------------------------------
# 항목 1: UID가 0(Root)인 일반 계정 존재 여부 점검
# --------------------------------------------------------------------
echo -n "[점검 1] UID가 0인 관리자급 계정 검사: "

# /etc/passwd 파일에서 3번째 열(UID)이 0인 계정의 이름(1번째 열)을 awk로 추출합니다.
# 기본 root 계정을 제외하고 다른 계정이 더 나오는지 확인합니다.
ROOT_ACCOUNTS=$(awk -F: '$3 == 0 {print $1}' /etc/passwd)

# 추출된 계정 수가 1개(즉, root만 존재)인지 확인합니다.
# echo "$ROOT_ACCOUNTS" | wc -l 명령어로 줄 수를 셉니다.
if [ "$(echo "$ROOT_ACCOUNTS" | wc -l)" -eq 1 ]; then
    echo -e "${GREEN}[안전] root 외에 UID가 0인 불법 계정이 없습니다.${NC}"
else
    echo -e "${RED}[취약] root 외에 UID가 0인 계정이 발견되었습니다!${NC}"
    echo "       -> 발견된 계정: $(echo $ROOT_ACCOUNTS | tr '\n' ' ')"
fi


# --------------------------------------------------------------------
# 항목 2: /etc/passwd 파일 권한 점검 (일반 유저 수정 불가능해야 함)
# --------------------------------------------------------------------
echo -n "[점검 2] /etc/passwd 파일 권한 점검: "

# stat 명령어를 사용해 파일의 권한(팔진수 형태, 예: 644)을 추출합니다.
FILE_PERM=$(stat -c "%a" /etc/passwd)

# /etc/passwd는 소유자만 수정 가능(6)하고 나머지는 읽기만(4) 가능한 644 이하여야 합니다.
# 안전한 기준인 644와 비교합니다. (-le는 Less than or Equal)
if [ "$FILE_PERM" -le 644 ]; then
    echo -e "${GREEN}[안전] /etc/passwd 파일 권한이 적절합니다. ($FILE_PERM)${NC}"
else
    echo -e "${RED}[취약] /etc/passwd 파일 권한이 과도합니다. ($FILE_PERM -> 644로 변경 필요)${NC}"
fi


# --------------------------------------------------------------------
# 항목 3: SSH 원격 접속 시 root 로그인 제한 여부 점검
# --------------------------------------------------------------------
echo -n "[점검 3] SSH root 직접 원격 접속 제한 여부: "

SSH_CFG="/etc/ssh/sshd_config"

# sshd_config 파일이 존재하는지 확인 (-f 옵션)
if [ -f "$SSH_CFG" ]; then
    # grep으로 PermitRootLogin 설정을 찾고, awk로 값(2번째 단어)만 추출합니다.
    # 주석처리(#)된 줄은 무시하기 위해 grep -v "^#"를 씁니다.
    ROOT_SSH_ALLOW=$(grep -i "PermitRootLogin" $SSH_CFG | grep -v "^#" | awk '{print $2}')
    
    # 설정 값이 no 이거나 prohibit-password 이면 안전하다고 판단합니다.
    if [ "$ROOT_SSH_ALLOW" = "no" ] || [ "$ROOT_SSH_ALLOW" = "prohibit-password" ]; then
        echo -e "${GREEN}[안전] root 직접 접속이 제한되어 있습니다. ($ROOT_SSH_ALLOW)${NC}"
    else
        echo -e "${RED}[취약] root 직접 접속이 허용되어 있을 수 있습니다. (설정값: $ROOT_SSH_ALLOW)${NC}"
    fi
else
    echo -e "${YELLOW}[경고] SSH 설정 파일($SSH_CFG)을 찾을 수 없습니다.${NC}"
fi


# --------------------------------------------------------------------
# 항목 4: 패스워드 없는 계정(비어 있는 패스워드) 존재 여부 점검
# --------------------------------------------------------------------
echo -n "[점검 4] 패스워드가 비어 있는 계정 검사: "

# /etc/shadow 파일의 2번째 열은 암호화된 패스워드 해시가 들어갑니다.
# 만약 이 칸이 비어있거나U(Ubuntu 특정 표기 등) 공백이라면 패스워드 없이 로그인이 가능함을 의미합니다.
# (점검을 위해 sudo 권한 필수)
EMPTY_PW_ACCOUNTS=$(sudo awk -F: '$2 == "" || $2 == "!" {next} $2 == "*" {next} length($2) < 5 {print $1}' /etc/shadow 2>/dev/null)

if [ -z "$EMPTY_PW_ACCOUNTS" ]; then
    echo -e "${GREEN}[안전] 패스워드가 비어 있는 계정이 없습니다.${NC}"
else
    echo -e "${RED}[취약] 패스워드 없이 로그인 가능한 계정이 존재합니다!${NC}"
    echo "       -> 위험 계정: $EMPTY_PW_ACCOUNTS"
fi

echo "----------------------------------------------------"
echo "점검이 완료되었습니다."
echo "====================================================="
